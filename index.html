PHP & MariaDB api/login.php

Your frontend submits JSON or POST fields (email, password) to api/login.php without altering your MariaDB schema: 

<?php
session_start();
header('Content-Type: application/json');
require_once '../db.php'; // PDO MariaDB connection

$data = json_decode(file_get_contents('php://input'), true) ?: $_POST;
$stmt = $pdo->prepare('SELECT id, name, password_hash FROM librarians WHERE email = ? LIMIT 1');
$stmt->execute([$data['email'] ?? '']);
$user = $stmt->fetch(PDO::FETCH_ASSOC);

if ($user && password_verify($data['password'] ?? '', $user['password_hash'])) {
    $_SESSION['librarian_id'] = $user['id'];
    echo json_encode(['ok' => true, 'name' => $user['name']]);
} else {
    http_response_code(401);
    echo json_encode(['error' => 'Invalid campus credentials']);
}
